    <div>

<script>
(function(){
  function getStore(key, fallback){
    try { return JSON.parse(localStorage.getItem(key)) || fallback; } catch(e){ return fallback; }
  }
  function setStore(key, value){
    try { localStorage.setItem(key, JSON.stringify(value)); } catch(e){}
  }
  function getEditableContent(base){
    var saved = getStore('roomContent.v1', {});
    return {
      resume: Object.assign({}, base.resume || {}, saved.resume || {}),
      books: Array.isArray(saved.books) ? saved.books : (base.books || []),
      music: Object.assign({}, base.music || {}, saved.music || {})
    };
  }
  function normalizeTracks(raw){
    if (Array.isArray(raw)) return raw.map(function(v){ return String(v).trim(); }).filter(Boolean);
    return String(raw || '').split(/[\s,，|]+/).map(function(v){ return v.trim(); }).filter(Boolean);
  }
  function esc(v){
    return String(v == null ? '' : v)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }
  function hashView(){
    var raw = decodeURIComponent((location.hash || '#resume').slice(1));
    return raw || 'resume';
  }
  function viewTitle(view){
    if (view.indexOf('photo:') === 0) return '照片';
    return {
      resume: '简历',
      photos: '照片墙',
      map: '地图',
      books: '书架',
      music: '歌曲',
      notebook: '桌上本子',
      manage: '内容管理'
    }[view] || '房间交互页';
  }
  function renderResume(content){
    var r = content.resume || {};
    return '<div class="hero">' +
      '<div class="card"><h2>' + esc(r.name || '我的简历') + '</h2><div class="meta">' + esc(r.title || '') + '</div>' +
      '<p>' + esc(r.bio || '') + '</p><div class="tags">' +
      (r.skills || []).map(function(s){ return '<span class="tag">' + esc(s) + '</span>'; }).join('') +
      '</div><div class="actions">' +
      (r.links || []).filter(function(l){ return l.url; }).map(function(l){ return '<a class="btn ghost" href="' + esc(l.url) + '" target="_blank" rel="noreferrer">' + esc(l.label) + '</a>'; }).join('') +
      '</div></div></div>';
  }
  function renderPhotos(view){
    var data = getStore('photoWall.v1', {});
    var photoIds = [];
    for (var i = 0; i < 12; i++) photoIds.push('p' + i);
    if (view.indexOf('photo:') === 0){
      var id = view.split(':')[1];
      var rec = data[id] || {};
      return '<div class="hero"><div class="actions"><a class="btn ghost" href="#photos">← 返回照片墙</a></div>' +
        (rec.img ? '<img src="' + rec.img + '" alt="photo">' : '<div class="card">这张照片还没有内容。</div>') +
        '<div class="card"><h2>照片 ' + esc(id.slice(1)) + '</h2><p>' + esc(rec.text || '还没有写说明。') + '</p></div></div>';
    }
    return '<div class="photo-grid">' + photoIds.map(function(id){
      var rec = data[id] || {};
      return '<a class="photo-card" href="#photo:' + esc(id) + '">' +
        (rec.img ? '<img src="' + rec.img + '" alt="photo">' : '<div class="empty">暂无照片</div>') +
        '<div class="copy">照片 ' + esc(id.slice(1)) + '<br>' + esc(rec.text || '点击查看') + '</div></a>';
    }).join('') + '</div>';
  }
  function renderMap(){
    var pins = getStore('mapPins.v1', []);
    var mapImage = getStore('mapImage.v1', './textures/tree-backdrop.jpg');
    return '<div class="hero"><div class="mapbox" id="mapbox"><img src="' + esc(mapImage) + '" alt="我的足迹地图">' +
      pins.map(function(pin){
        var x = Math.max(0, Math.min(1, Number(pin.x) || 0.5)) * 100;
        var y = Math.max(0, Math.min(1, Number(pin.y) || 0.5)) * 100;
        return '<span class="pin" title="' + esc(pin.title || '未命名地点') + '" style="left:' + x + '%;top:' + y + '%"></span>';
      }).join('') + '</div>' +
      '<div class="card"><h2>我的足迹地图</h2><p>已记录 ' + pins.length + ' 个图钉。</p><div class="meta">' +
      pins.map(function(pin){ return esc(pin.title || '未命名地点') + (pin.text ? '：' + esc(pin.text) : ''); }).join('<br>') +
      (pins.length ? '' : '还没有添加图钉。') + '</div></div></div>';
  }
  function renderBooks(content){
    var books = content.books || [];
    return '<div class="book-list">' + books.map(function(b){
      return '<a class="book" href="' + esc(b.link || '#') + '" target="_blank" rel="noreferrer">' +
        '<div class="spine" style="background:' + esc(b.color || '#8a6a48') + '">' + esc(b.title) + '</div>' +
        '<div class="copy">' + esc(b.author || '') + '</div></a>';
    }).join('') + '</div>';
  }
  function renderMusic(content){
    var conf = getStore('music.v2', null) || getStore('music.v1', content.music || {}) || {};
    var pid = conf.playlistId || '';
    var tracks = normalizeTracks(conf.tracks || conf.trackIds || conf.songIds || '');
    var currentIndex = parseInt(conf.currentIndex, 10);
    if (!isFinite(currentIndex)) currentIndex = 0;
    var currentTrack = tracks.length ? tracks[Math.min(currentIndex, tracks.length - 1)] : '';
    return '<div class="hero music">' +
      (currentTrack ? '<iframe src="https://music.163.com/outchain/player?type=2&id=' + esc(currentTrack) + '&auto=0&height=430"></iframe>' :
      (pid ? '<iframe src="https://music.163.com/outchain/player?type=0&id=' + esc(pid) + '&auto=0&height=430"></iframe>' :
      '<div class="card"><h2>还没绑定网易云内容</h2><p>推荐填写“歌曲 ID 列表”，这样点击房间里的唱片机会直接切到下一首。</p></div>')) +
      '<div class="card stack"><h2>网易云配置</h2>' +
      '<div><div class="meta">歌单 ID（可选）</div><input id="pid" type="text" placeholder="歌单 id，例如 24381616" value="' + esc(pid) + '"></div>' +
      '<div><div class="meta">歌曲 ID 列表（推荐）</div><textarea id="trackIds" placeholder="填多个歌曲 id，用空格、逗号或换行分隔。点唱片机时会按顺序切歌。">' + esc(tracks.join('\n')) + '</textarea></div>' +
      '<div class="meta">当前共 ' + tracks.length + ' 首' + (currentTrack ? '，预览的是第 ' + (Math.min(currentIndex, tracks.length - 1) + 1) + ' 首。' : '。') + '</div>' +
      '<div class="actions"><button id="savePid">保存配置</button></div></div></div>';
  }
  function renderNotebook(){
    var note = getStore('deskNote.v1', { title: '桌上本子', text: '' });
    return '<div class="hero"><div class="card"><h2>' + esc(note.title || '桌上本子') + '</h2><textarea id="notebookText" placeholder="写点什么…">' + esc(note.text || '') + '</textarea><div class="actions"><button id="saveNote">保存本子内容</button></div></div></div>';
  }
  function renderManage(content){
    var r = content.resume || {};
    var books = content.books || [];
    var pins = getStore('mapPins.v1', []);
    var mapImage = getStore('mapImage.v1', '');
    var note = getStore('deskNote.v1', { title: '桌上本子', text: '' });
    var music = getStore('music.v2', null) || getStore('music.v1', content.music || {}) || {};
    var tracks = normalizeTracks(music.tracks || music.trackIds || music.songIds || '');
    var photos = [];
    for (var i = 0; i < 12; i++) photos.push('p' + i);
    return '<div class="hero"><div class="notice">所有内容会保存在当前浏览器的本地存储中，不需要上传服务器。换电脑或清除浏览器数据后，需要重新导入；正式发布前可把内容同步回项目文件。</div>' +
      '<div class="editor-grid">' +
      '<section class="editor-card"><h3>Resume / 简历</h3>' +
      '<label>姓名</label><input id="resumeName" value="' + esc(r.name || '') + '">' +
      '<label>身份/标题</label><input id="resumeTitle" value="' + esc(r.title || '') + '">' +
      '<label>个人简介</label><textarea id="resumeBio">' + esc(r.bio || '') + '</textarea>' +
      '<label>技能（用逗号分隔）</label><input id="resumeSkills" value="' + esc((r.skills || []).join(', ')) + '">' +
      '<label>链接 JSON</label><textarea id="resumeLinks">' + esc(JSON.stringify(r.links || [], null, 2)) + '</textarea></section>' +
      '<section class="editor-card"><h3>Books / 书架</h3><label>书籍列表 JSON</label><textarea id="booksJson">' + esc(JSON.stringify(books, null, 2)) + '</textarea><div class="meta">每本书使用 title、author、color、link 字段。</div></section>' +
      '<section class="editor-card"><h3>Photos / 照片墙</h3><div class="photo-editor">' +
      photos.map(function(id){
        var rec = getStore('photoWall.v1', {})[id] || {};
        return '<div class="slot"><strong>' + id + '</strong>' + (rec.img ? '<img src="' + esc(rec.img) + '" alt="' + id + '">' : '<div class="empty">暂无图片</div>') +
          '<input type="file" accept="image/*" data-photo-file="' + id + '">' +
          '<input data-photo-text="' + id + '" placeholder="照片说明" value="' + esc(rec.text || '') + '"></div>';
      }).join('') + '</div></section>' +
      '<section class="editor-card"><h3>Map / 地图</h3><label>上传地图底图</label><input id="mapFile" type="file" accept="image/*">' +
      (mapImage ? '<img src="' + esc(mapImage) + '" alt="当前地图" style="width:100%;max-height:180px;object-fit:cover;border-radius:10px;margin-top:8px">' : '') +
      '<label>图钉 JSON</label><textarea id="pinsJson">' + esc(JSON.stringify(pins, null, 2)) + '</textarea><div class="meta">每个图钉使用 x、y、title、text 字段，x/y 为 0 到 1 的百分比。</div></section>' +
      '<section class="editor-card"><h3>Notebook / 本子</h3><label>标题</label><input id="noteTitle" value="' + esc(note.title || '桌上本子') + '">' +
      '<label>内容</label><textarea id="noteText">' + esc(note.text || '') + '</textarea></section>' +
      '<section class="editor-card"><h3>Music / 网易云</h3><label>歌单 ID（可选）</label><input id="musicPlaylist" value="' + esc(music.playlistId || '') + '">' +
      '<label>歌曲 ID（每行一个）</label><textarea id="musicTracks">' + esc(tracks.join('\n')) + '</textarea></section>' +
      '</div><div class="actions"><button id="saveAll">保存全部内容</button><a class="btn ghost" href="./index.html">回到房间测试</a></div></div>';
  }
  function readFileAsDataUrl(file, done){
    if (!file) return done('');
    var reader = new FileReader();
    reader.onload = function(){ done(reader.result); };
    reader.readAsDataURL(file);
  }
  function bindManage(content){
    var saveAll = document.getElementById('saveAll');
    if (!saveAll) return;
    var pendingPhotos = {};
    document.querySelectorAll('[data-photo-file]').forEach(function(input){
      input.onchange = function(){
        readFileAsDataUrl(input.files && input.files[0], function(data){ pendingPhotos[input.dataset.photoFile] = data; });
      };
    });
    var mapData = '';
    var mapFile = document.getElementById('mapFile');
    if (mapFile) mapFile.onchange = function(){ readFileAsDataUrl(mapFile.files && mapFile.files[0], function(data){ mapData = data; }); };
    saveAll.onclick = function(){
      var links = [];
      try { links = JSON.parse(document.getElementById('resumeLinks').value || '[]'); } catch(e){ alert('Resume 链接 JSON 格式不正确'); return; }
      var books;
      try { books = JSON.parse(document.getElementById('booksJson').value || '[]'); } catch(e){ alert('书籍 JSON 格式不正确'); return; }
      var pins;
      try { pins = JSON.parse(document.getElementById('pinsJson').value || '[]'); } catch(e){ alert('地图图钉 JSON 格式不正确'); return; }
      var next = getEditableContent(content);
      next.resume = {
        name: document.getElementById('resumeName').value.trim(),
        title: document.getElementById('resumeTitle').value.trim(),
        bio: document.getElementById('resumeBio').value,
        skills: document.getElementById('resumeSkills').value.split(/[,，]/).map(function(v){ return v.trim(); }).filter(Boolean),
        links: links
      };
      next.books = books;
      next.music = { playlistId: document.getElementById('musicPlaylist').value.trim() };
      setStore('roomContent.v1', next);
      setStore('mapPins.v1', pins);
      setStore('deskNote.v1', { title: document.getElementById('noteTitle').value.trim() || '桌上本子', text: document.getElementById('noteText').value });
      setStore('music.v2', { playlistId: next.music.playlistId, tracks: normalizeTracks(document.getElementById('musicTracks').value), currentIndex: -1 });
      var oldPhotos = getStore('photoWall.v1', {});
      document.querySelectorAll('[data-photo-text]').forEach(function(input){
        var id = input.dataset.photoText;
        oldPhotos[id] = oldPhotos[id] || {};
        oldPhotos[id].text = input.value;
        if (pendingPhotos[id]) oldPhotos[id].img = pendingPhotos[id];
      });
      setStore('photoWall.v1', oldPhotos);
      if (mapData) setStore('mapImage.v1', mapData);
      saveAll.textContent = '已保存 ✓';
    };
  }
  function mount(content){
    var view = hashView();
    document.getElementById('pageTitle').textContent = viewTitle(view);
    document.getElementById('pageDesc').textContent = '当前页面：' + viewTitle(view);
    var app = document.getElementById('app');
    if (view === 'manage') app.innerHTML = renderManage(content);
    else if (view === 'resume') app.innerHTML = renderResume(content);
    else if (view === 'photos' || view.indexOf('photo:') === 0) app.innerHTML = renderPhotos(view);
    else if (view === 'map') app.innerHTML = renderMap();
    else if (view === 'books') app.innerHTML = renderBooks(content);
    else if (view === 'music') app.innerHTML = renderMusic(content);
    else if (view === 'notebook') app.innerHTML = renderNotebook();
    else app.innerHTML = '<div class="card">暂未实现这个页面。</div>';

    var savePid = document.getElementById('savePid');
    if (savePid) savePid.onclick = function(){
      setStore('music.v2', {
        playlistId: document.getElementById('pid').value.trim(),
        tracks: normalizeTracks(document.getElementById('trackIds').value),
        currentIndex: 0
      });
      mount(content);
    };
    var saveNote = document.getElementById('saveNote');
    if (saveNote) saveNote.onclick = function(){
      var note = getStore('deskNote.v1', { title: '桌上本子', text: '' });
      note.text = document.getElementById('notebookText').value;
      setStore('deskNote.v1', note);
      saveNote.textContent = '已保存 ✓';
    };
    if (view === 'manage') bindManage(content);
  }

  var content = { resume: {}, books: [], music: {} };
  fetch('./content.json').then(function(r){ return r.ok ? r.json() : null; }).then(function(j){
    if (j) content = getEditableContent(j);
    mount(content);
  }).catch(function(){
    content = getEditableContent(content);
    mount(content);
  });
  window.addEventListener('hashchange', function(){ mount(content); });
})();
</script>
